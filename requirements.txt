import numpy as np
import matplotlib.pyplot as plt

# Configuração do gerador estocástico
rng = np.random.default_rng(42)
n_simulacoes = 100000

print(f"Executando {n_simulacoes:,} simulações para mapear a distribuição de somas...")

# Coleta das somas das combinações geradas aleatoriamente
somas = [sum(rng.choice(range(1, 61), size=6, replace=False)) for _ in range(n_simulacoes)]

# Configuração e plotagem do gráfico
plt.figure(figsize=(11, 6))

# Histograma de frequências
n, bins, patches = plt.hist(somas, bins=45, color='royalblue', edgecolor='navy', alpha=0.75)

# Linhas de referência dos filtros paramétricos
plt.axvline(x=130, color='darkorange', linestyle='--', linewidth=2, label='Limite Inferior (130)')
plt.axvline(x=220, color='crimson', linestyle='--', linewidth=2, label='Limite Superior (220)')

# Detalhes visuais e estatísticos
plt.title('Distribuição de Frequência das Somas (Simulação de Monte Carlo)', fontsize=12, fontweight='bold', pad=15)
plt.xlabel('Valor da Soma das 6 Dezenas', fontsize=10, fontweight='bold')
plt.ylabel('Frequência Absoluta', fontsize=10, fontweight='bold')
plt.legend(loc='upper right', frameon=True)
plt.grid(axis='y', linestyle=':', alpha=0.7)

# Anotação informativa no gráfico
media_soma = np.mean(somas)
plt.annotate(f'Média Teórica: {media_soma:.1f}', 
             xy=(media_soma, max(n)*0.5), 
             xytext=(media_soma + 15, max(n)*0.6),
             arrowprops=dict(facecolor='black', shrink=0.05, width=1, headwidth=6),
             fontsize=9, fontweight='bold')

plt.tight_layout()
